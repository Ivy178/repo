# 添加Elastic Helm仓库
helm repo add elastic https://helm.elastic.co
helm repo update

# elasticsearch
> 部署Elasticsearch集群（生产环境至少3节点）
helm install elasticsearch elastic/elasticsearch \
  --namespace logging \
  --create-namespace \
  --set replicas=3 \
  --set resources.requests.memory=4Gi \
  --set resources.requests.cpu=1

> 验证 Elasticsearch 服务
kubectl get pods -l app=elasticsearch
kubectl get service elasticsearch-master



# filebeat
> 部署Filebeat DaemonSet
helm install filebeat elastic/filebeat \
  --namespace logging \
  --set daemonset.enabled=true

> 验证 Filebeat
kubectl get pods -l app=filebeat


# Kibana
> 部署Kibana
helm install kibana elastic/kibana \  
  --set service.type=NodePort \  
  --set elasticsearchHosts=http://elasticsearch-master:9200     // service name 和 端酒


> 验证 Kibana：
kubectl get pods -l app=kibana
kubectl get svc -l app=kibana
